## 功能模块

### 📖 模块一：PDF读取

#### 提取全部文本
```python
import fitz

doc = fitz.open("file.pdf")
for page in doc:
    text = page.get_text()
    print(text)
doc.close()
```

#### 提取特定区域
```python
page = doc[0]
# 定义矩形区域 (x0, y0, x1, y1)
rect = fitz.Rect(100, 100, 500, 500)
text = page.get_text("text", clip=rect)
```

#### 提取图片
```python
for page_num in range(len(doc)):
    page = doc[page_num]
    images = page.get_images()
    for img_index, img in enumerate(images):
        xref = img[0]
        pix = fitz.Pixmap(doc, xref)
        pix.save(f"image_{page_num}_{img_index}.png")
```

#### 提取表格
```python
import fitz

doc = fitz.open("file.pdf")
page = doc[0]
tabs = page.find_tables()

for tab in tabs:
    # 提取表格数据
    df = tab.to_pandas()  # 转为DataFrame
    print(df)
```

### 🔀 模块三：PDF合并拆分

#### 合并多个PDF
```python
import fitz

# 创建新文档
result = fitz.open()

# 合并多个PDF
pdf_files = ["file1.pdf", "file2.pdf", "file3.pdf"]
for pdf_file in pdf_files:
    doc = fitz.open(pdf_file)
    result.insert_pdf(doc)
    doc.close()

result.save("merged.pdf")
result.close()
```

#### 拆分PDF
```python
import fitz

doc = fitz.open("original.pdf")

# 拆分为单页PDF
for i in range(len(doc)):
    new_doc = fitz.open()
    new_doc.insert_pdf(doc, from_page=i, to_page=i)
    new_doc.save(f"page_{i+1}.pdf")
    new_doc.close()

doc.close()
```

#### 提取特定页面范围
```python
doc = fitz.open("original.pdf")
new_doc = fitz.open()
new_doc.insert_pdf(doc, from_page=5, to_page=10)  # 提取第6-11页
new_doc.save("extracted.pdf")
```

### 🔒 模块四：PDF加密解密

#### 加密PDF
```python
doc = fitz.open("file.pdf")

# 设置密码保护
perm = int(
    fitz.PDF_PERM_ACCESSIBILITY  # 允许辅助功能
    | fitz.PDF_PERM_PRINT  # 允许打印
)
owner_pw = "owner123"  # 所有者密码
user_pw = "user123"    # 用户密码

doc.save("encrypted.pdf", encryption=fitz.PDF_ENCRYPT_AES_256, owner_pw=owner_pw, user_pw=user_pw, permissions=perm)
```

#### 解密PDF
```python
doc = fitz.open("encrypted.pdf")
if doc.is_encrypted:
    doc.authenticate("user123")  # 输入密码
    # 现在可以访问内容了
```

### 📝 模块五：PDF生成

#### 从零创建PDF
```python
import fitz

doc = fitz.open()  # 创建空白文档
page = doc.new_page()

# 添加文本
page.insert_text(fitz.Point(100, 100), "Hello World", fontsize=24)

# 添加图片
page.insert_image(fitz.Rect(100, 200, 300, 400), filename="image.png")

# 添加矩形
shape = page.new_shape()
shape.draw_rect(fitz.Rect(100, 100, 400, 400))
shape.finish(color=(1, 0, 0), width=2)

doc.save("created.pdf")
```

#### 图片转PDF
```python
import fitz

doc = fitz.open()
images = ["img1.jpg", "img2.png", "img3.jpg"]

for img in images:
    page = doc.new_page()
    img_rect = fitz.Rect(0, 0, 612, 792)  # US Letter size
    page.insert_image(img_rect, filename=img)

doc.save("images_to_pdf.pdf")
```

#### 文本转PDF
```python
import fitz

doc = fitz.open()
page = doc.new_page()

text = """
这是第一段文字。
这是第二段文字。
"""

# 插入多行文本
text_rect = fitz.Rect(50, 50, 550, 750)
page.insert_textbox(text_rect, text, fontsize=12, fontname="helv")

doc.save("text_to_pdf.pdf")
```

## 高级功能示例

### 批量添加水印
```python
import fitz
import os

def add_watermark_batch(pdf_dir, watermark_text, output_dir):
    """批量为PDF添加水印"""
    for filename in os.listdir(pdf_dir):
        if filename.endswith('.pdf'):
            doc = fitz.open(os.path.join(pdf_dir, filename))

            for page in doc:
                page.insert_text(
                    fitz.Point(200, 400),
                    watermark_text,
                    fontsize=60,
                    color=(0.9, 0.9, 0.9),
                    rotate=45
                )

            doc.save(os.path.join(output_dir, f"watermarked_{filename}"))
            doc.close()
```

### 自动化PDF报告生成
```python
import fitz
from datetime import datetime

def generate_report(title, content, output_file):
    """生成格式化PDF报告"""
    doc = fitz.open()
    page = doc.new_page()

    # 添加标题
    page.insert_text(
        fitz.Point(100, 100),
        title,
        fontsize=24,
        fontname="helv-b"
    )

    # 添加日期
    date = datetime.now().strftime("%Y-%m-%d")
    page.insert_text(
        fitz.Point(100, 150),
        f"生成日期: {date}",
        fontsize=12
    )

    # 添加内容
    text_rect = fitz.Rect(100, 200, 512, 750)
    page.insert_textbox(text_rect, content, fontsize=11)

    # 添加页脚
    page.insert_text(
        fitz.Point(250, 750),
        "第 1 页",
        fontsize=10
    )

    doc.save(output_file)
    doc.close()
```

## 重要提示

- **性能优化**：处理大文件时使用`page.get_text("dict")`获取结构化数据
- **内存管理**：处理完记得`doc.close()`
- **异常处理**：始终用try-except包裹文件操作

## 完整示例脚本

```python
#!/usr/bin/env python3
"""
PDF处理工具集 - 基于PyMuPDF
支持：读取、编辑、合并、拆分、加密、生成
"""
import fitz
import sys

class PDFTool:
    def __init__(self, pdf_path):
        self.doc = fitz.open(pdf_path)

    def extract_text(self, page_num=None):
        """提取文本"""
        if page_num is not None:
            return self.doc[page_num].get_text()
        return [page.get_text() for page in self.doc]

    def add_watermark(self, text, output_path):
        """添加水印"""
        for page in self.doc:
            page.insert_text(
                fitz.Point(200, 400),
                text,
                fontsize=60,
                color=(0.8, 0.8, 0.8),
                rotate=45
            )
        self.doc.save(output_path)

    def split(self, output_dir):
        """拆分为单页"""
        for i in range(len(self.doc)):
            new_doc = fitz.open()
            new_doc.insert_pdf(self.doc, from_page=i, to_page=i)
            new_doc.save(f"{output_dir}/page_{i+1}.pdf")
            new_doc.close()

    def close(self):
        self.doc.close()

if __name__ == "__main__":
    # 使用示例
    tool = PDFTool("example.pdf")
    print(tool.extract_text())
    tool.add_watermark("CONFIDENTIAL", "output.pdf")
    tool.close()
```
